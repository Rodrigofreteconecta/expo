// freteconecta_app - Chat com Foto, Vídeo, Áudio, Notificações Push e Ícone de Sino com Contador import React, { useState, useEffect, useRef } from 'react'; import { View, Text, TextInput, TouchableOpacity, ScrollView, Platform, Linking, Image, Alert } from 'react-native'; import { Video } from 'expo-av'; import * as DocumentPicker from 'expo-document-picker'; import * as ImagePicker from 'expo-image-picker'; import { Audio } from 'expo-av'; import * as Notifications from 'expo-notifications'; import * as Device from 'expo-device'; import { Bell } from 'lucide-react-native';

Notifications.setNotificationHandler({ handleNotification: async () => ({ shouldShowAlert: true, shouldPlaySound: true, shouldSetBadge: false, }), });

export default function App() { const [screen, setScreen] = useState('login'); const [historico, setHistorico] = useState([]); const [publicacoes, setPublicacoes] = useState([]); const [tipoUsuario, setTipoUsuario] = useState('motorista'); const [chatPorCarga, setChatPorCarga] = useState({}); const [cargaSelecionada, setCargaSelecionada] = useState(null); const [notificacoes, setNotificacoes] = useState([]); const [notificacoesNaoLidas, setNotificacoesNaoLidas] = useState(0); const notificationListener = useRef(); const responseListener = useRef();

useEffect(() => { registerForPushNotificationsAsync();

notificationListener.current = Notifications.addNotificationReceivedListener(notification => {
  console.log('Notificação recebida:', notification);
  setNotificacoes(prev => [...prev, notification.request.content]);
  setNotificacoesNaoLidas(prev => prev + 1);
});

responseListener.current = Notifications.addNotificationResponseReceivedListener(response => {
  console.log('Resposta à notificação:', response);
});

return () => {
  Notifications.removeNotificationSubscription(notificationListener.current);
  Notifications.removeNotificationSubscription(responseListener.current);
};

}, []);

function adicionarPublicacao(carga) { setPublicacoes([...publicacoes, carga]); enviarNotificacao('Nova carga publicada', 'Uma nova carga está disponível para transporte.'); }

return ( <View className="flex-1 bg-white"> <View className="flex-row justify-between items-center px-4 py-2 bg-gray-100"> <Text className="text-lg font-bold">FreteConecta</Text> <TouchableOpacity onPress={() => setScreen('notificacoes')}> <Bell size={24} color="black" /> {notificacoesNaoLidas > 0 && ( <View className="absolute top-0 right-0 bg-red-500 text-white text-xs p-1 rounded-full"> <Text>{notificacoesNaoLidas}</Text> </View> )} </TouchableOpacity> </View>

{screen === 'login' && <LoginScreen navigate={setScreen} setTipoUsuario={setTipoUsuario} />}
  {screen === 'cadastro' && <CadastroScreen navigate={setScreen} setTipoUsuario={setTipoUsuario} />}
  {screen === 'publicar' && <PublicarCargaScreen navigate={setScreen} adicionarPublicacao={adicionarPublicacao} />}
  {screen === 'listar' && <ListaCargasScreen navigate={setScreen} cargas={publicacoes} selecionarCarga={(c) => { setCargaSelecionada(c); setScreen('chat'); }} />}
  {screen === 'historico' && <HistoricoScreen navigate={setScreen} historico={historico} />}
  {screen === 'minhasPublicacoes' && <MinhasPublicacoesScreen navigate={setScreen} publicacoes={publicacoes} abrirChat={(c) => { setCargaSelecionada(c); setScreen('chat'); }} />}
  {screen === 'chat' && <ChatScreen carga={cargaSelecionada} mensagens={chatPorCarga[cargaSelecionada?.origem + cargaSelecionada?.destino] || []} enviarMensagem={(msg) => {
    const chave = cargaSelecionada.origem + cargaSelecionada.destino;
    const novas = [...(chatPorCarga[chave] || []), msg];
    setChatPorCarga({ ...chatPorCarga, [chave]: novas });
    enviarNotificacao('Nova mensagem', 'Você recebeu uma nova mensagem.');
  }} navigate={setScreen} tipoUsuario={tipoUsuario} />}
  {screen === 'notificacoes' && <NotificacoesScreen notificacoes={notificacoes} navigate={setScreen} />}
</View>

); }

function NotificacoesScreen({ notificacoes, navigate }) { return ( <View className="flex-1 bg-white p-4"> <Text className="text-xl font-bold mb-4">Notificações</Text> <ScrollView> {notificacoes.map((n, index) => ( <View key={index} className="mb-3 p-3 bg-gray-100 rounded-lg"> <Text className="font-semibold">{n.title}</Text> <Text>{n.body}</Text> </View> ))} </ScrollView> <TouchableOpacity onPress={() => navigate('listar')} className="mt-4 bg-blue-500 p-3 rounded-lg"> <Text className="text-white text-center">Voltar</Text> </TouchableOpacity> </View> ); }

async function registerForPushNotificationsAsync() { if (Device.isDevice) { const { status: existingStatus } = await Notifications.getPermissionsAsync(); let finalStatus = existingStatus; if (existingStatus !== 'granted') { const { status } = await Notifications.requestPermissionsAsync(); finalStatus = status; } if (finalStatus !== 'granted') { Alert.alert('Erro', 'Permissão para notificações não concedida.'); return; } await Notifications.getExpoPushTokenAsync(); } else { Alert.alert('Erro', 'As notificações só funcionam em dispositivos físicos.'); } }

async function enviarNotificacao(titulo, corpo) { await Notifications.scheduleNotificationAsync({ content: { title: titulo, body: corpo, }, trigger: null, }); }
