// FreteConecta App - Versão sem notificacoes para Snack import React, { useState } from 'react'; import { View, Text, TouchableOpacity, ScrollView, TextInput, Image } from 'react-native'; import { Video } from 'expo-av'; import * as DocumentPicker from 'expo-document-picker'; import * as ImagePicker from 'expo-image-picker'; import { Audio } from 'expo-av'; import { Bell } from 'lucide-react-native';

export default function App() { const [screen, setScreen] = useState('login'); const [historico, setHistorico] = useState([]); const [publicacoes, setPublicacoes] = useState([]); const [tipoUsuario, setTipoUsuario] = useState('motorista'); const [chatPorCarga, setChatPorCarga] = useState({}); const [cargaSelecionada, setCargaSelecionada] = useState(null); const [notificacoes, setNotificacoes] = useState([]); const [notificacoesNaoLidas, setNotificacoesNaoLidas] = useState(0);

function adicionarPublicacao(carga) { setPublicacoes([...publicacoes, carga]); // Notificacao removida para funcionar no Snack }

return ( <View style={{ padding: 20, marginTop: 40 }}> <Text style={{ fontSize: 24, fontWeight: 'bold' }}>FreteConecta</Text>

<TouchableOpacity onPress={() => setScreen('notificacoes')}>
    <Bell />
    {notificacoesNaoLidas > 0 && (
      <Text style={{ color: 'red' }}>{notificacoesNaoLidas}</Text>
    )}
  </TouchableOpacity>

  {screen === 'login' && <LoginScreen navigate={setScreen} setTipoUsuario={setTipoUsuario} />}
  {screen === 'cadastro' && <CadastroScreen navigate={setScreen} setTipoUsuario={setTipoUsuario} />}
  {screen === 'publicar' && <PublicarCargaScreen navigate={setScreen} adicionarPublicacao={adicionarPublicacao} />}
  {screen === 'listar' && <ListaCargasScreen navigate={setScreen} cargas={publicacoes} selecionarCarga={(c) => { setCargaSelecionada(c); setScreen('chat'); }} />}
  {screen === 'historico' && <HistoricoScreen navigate={setScreen} historico={historico} />}
  {screen === 'minhasPublicacoes' && <MinhasPublicacoesScreen navigate={setScreen} publicacoes={publicacoes} abrirChat={(c) => { setCargaSelecionada(c); setScreen('chat'); }} />}
  {screen === 'chat' && <ChatScreen carga={cargaSelecionada} mensagens={chatPorCarga[cargaSelecionada?.origem + cargaSelecionada?.destino] || []} enviarMensagem={(msg) => {
    const chave = cargaSelecionada.origem + cargaSelecionada.destino;
    const novas = [...(chatPorCarga[chave] || []), msg];
    setChatPorCarga({ ...chatPorCarga, [chave]: novas });
  }} navigate={setScreen} tipoUsuario={tipoUsuario} />}
  {screen === 'notificacoes' && <NotificacoesScreen notificacoes={notificacoes} navigate={setScreen} />}
</View>

); }

function LoginScreen({ navigate, setTipoUsuario }) { return ( <View> <Text>Login</Text> <TouchableOpacity onPress={() => navigate('listar')}><Text>Entrar</Text></TouchableOpacity> </View> ); }

function CadastroScreen({ navigate, setTipoUsuario }) { return ( <View> <Text>Cadastro</Text> <TouchableOpacity onPress={() => navigate('login')}><Text>Voltar</Text></TouchableOpacity> </View> ); }

function PublicarCargaScreen({ navigate, adicionarPublicacao }) { return ( <View> <Text>Publicar Carga</Text> <TouchableOpacity onPress={() => { adicionarPublicacao({ origem: 'SP', destino: 'RJ', descricao: 'Carga teste' }); navigate('listar'); }}><Text>Publicar</Text></TouchableOpacity> </View> ); }

function ListaCargasScreen({ cargas, selecionarCarga, navigate }) { return ( <ScrollView> {cargas.map((c, i) => ( <TouchableOpacity key={i} onPress={() => selecionarCarga(c)}> <Text>{c.origem} - {c.destino}</Text> </TouchableOpacity> ))} </ScrollView> ); }

function ChatScreen({ carga, mensagens, enviarMensagem, navigate, tipoUsuario }) { return ( <View> <Text>Chat com {carga?.origem} → {carga?.destino}</Text> {mensagens.map((msg, i) => ( <Text key={i}>{msg}</Text> ))} <TouchableOpacity onPress={() => enviarMensagem('Olá!')}><Text>Enviar "Olá!"</Text></TouchableOpacity> </View> ); }

function NotificacoesScreen({ notificacoes, navigate }) { return ( <View> <Text>Notificações</Text> {notificacoes.map((n, index) => ( <View key={index}> <Text>{n.title}</Text> <Text>{n.body}</Text> </View> ))} <TouchableOpacity onPress={() => navigate('listar')}><Text>Voltar</Text></TouchableOpacity> </View> ); }

function HistoricoScreen({ historico, navigate }) { return ( <View> <Text>Histórico</Text> </View> ); }

function MinhasPublicacoesScreen({ publicacoes, abrirChat, navigate }) { return ( <ScrollView> {publicacoes.map((p, i) => ( <TouchableOpacity key={i} onPress={() => abrirChat(p)}> <Text>{p.origem} → {p.destino}</Text> </TouchableOpacity> ))} </ScrollView> ); }
